A minimal blog: MDX posts, light list on the index, full document on the detail page.

## Install

```package-install
@anhur/core @anhur/vite @anhur/mdx
```

## Config

```ts title="anhur.config.ts"
import { defineCollection, defineConfig, schema as s } from "@anhur/core";
import { mdx, schema as m } from "@anhur/mdx";

const posts = defineCollection({
  name: "posts",
  directory: "content/posts",
  include: "**/*.mdx",
  localized: false,
  generate: {
    split: "light",
    listSort: { by: "publishedAt", order: "desc" },
  },
  schema: s.object({
    title: s.string(),
    slug: s.unique(),
    publishedAt: s.isodate(),
    body: m.mdx(),
  }),
});

export default defineConfig({
  processors: [mdx({ gfm: true })],
  content: [posts],
});
```

## Vite

```ts title="vite.config.ts"
import { defineConfig } from "vite";
import anhur from "@anhur/vite";

export default defineConfig({
  plugins: [anhur()],
});
```


## App usage

```ts
import { allPosts, getPost } from "anhur/generated";
import { MDXContent } from "@anhur/mdx/react";

export function BlogIndex() {
  return (
    <ul>
      {allPosts.map((post) => (
        <li key={post.slug}>
          <a href={`/blog/${post.slug}`}>{post.title}</a>
        </li>
      ))}
    </ul>
  );
}

export async function BlogPost({ slug }: { slug: string }) {
  const post = await getPost({ slug });
  if (!post) return null;
  return (
    <article>
      <h1>{post.title}</h1>
      <MDXContent code={post.body} />
    </article>
  );
}
```
