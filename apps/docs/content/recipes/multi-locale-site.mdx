English and German posts plus localized site settings.

## Install

```package-install
@anhur/core @anhur/vite @anhur/markdown
```

## Config

```ts title="anhur.config.ts"
import { defineCollection, defineConfig, defineSingleton, schema as s } from "@anhur/core";
import { markdown, schema as md } from "@anhur/markdown";

const posts = defineCollection({
  name: "posts",
  directory: "content/posts",
  include: "**/*.md",
  schema: s.object({
    title: s.string(),
    slug: s.unique(),
    body: md.markdown(),
  }),
});

const settings = defineSingleton({
  name: "settings",
  directory: "content/settings",
  schema: s.object({
    siteTitle: s.string(),
  }),
});

export default defineConfig({
  localization: {
    strategy: "folder",
    locales: ["en", "de"],
    defaultLocale: "en",
  },
  processors: [markdown({ gfm: true })],
  content: [posts, settings],
});
```

## Files

```txt
content/posts/en/hello.md
content/posts/de/hello.md
content/settings/en/index.md
content/settings/de/index.md
```

```md title="content/posts/en/hello.md"
---
title: Hello
slug: hello
---

English body.
```

```md title="content/posts/de/hello.md"
---
title: Hallo
slug: hello
---

Deutscher Text.
```

## Usage

```ts
import { allPosts, getPost, settings, getSettings } from "anhur/generated";

const german = allPosts.filter((p) => p._meta.locale === "de");
const post = await getPost({ locale: "de", slug: "hello" });
const deSettings = await getSettings({ locale: "de" });

console.log(settings.siteTitle); // default locale
```

Exact singleton export names depend on `generate` defaults — adjust if you rename them.
